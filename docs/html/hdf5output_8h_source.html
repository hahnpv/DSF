97 void createFile()
98 {
99 if (file_ready || deferred_filename.empty()) return;
100 try {
101 file = new H5::H5File(deferred_filename, H5F_ACC_TRUNC);
102
103 // ---- Write root-level metadata attributes ----
104 writeRootMetadata();
105
106 // Create groups for all unique group names
107 for (const auto& gname : used_groups) {
108 if (!gname.empty() && groups.find(gname) == groups.end()) {
109 groups[gname] = new H5::Group(file->createGroup(gname));
110 }
111 }
112
113 // Create Time dataset at file root
114 createDatasetAt("Time", H5::PredType::NATIVE_DOUBLE, nullptr, "s", "Time", 1.0);
115
116 // Create datasets for all registered variables
117 for (size_t i = 0; i < titles.size(); i++)
118 createDatasetAt(titles[i], H5::PredType::NATIVE_DOUBLE,
119 groupFor(var_groups[i]),
120 unit_strings[i], titles[i], conversions[i]);
121
122 for (size_t i = 0; i < vec_titles.size(); i++) {
123 H5::Group* g = groupFor(vec_var_groups[i]);
124 createDatasetAt(vec_titles[i] + "_x", H5::PredType::NATIVE_DOUBLE, g,
125 vec_units[i], vec_titles[i] + " (x)", vec_conversions[i]);
126 createDatasetAt(vec_titles[i] + "_y", H5::PredType::NATIVE_DOUBLE, g,
127 vec_units[i], vec_titles[i] + " (y)", vec_conversions[i]);
128 createDatasetAt(vec_titles[i] + "_z", H5::PredType::NATIVE_DOUBLE, g,
129 vec_units[i], vec_titles[i] + " (z)", vec_conversions[i]);
130 }
131
132 for (size_t i = 0; i < quat_titles.size(); i++) {
133 H5::Group* g = groupFor(quat_var_groups[i]);
134 createDatasetAt(quat_titles[i] + "_x", H5::PredType::NATIVE_DOUBLE, g,
135 quat_units[i], quat_titles[i] + " (x)", quat_conversions[i]);
136 createDatasetAt(quat_titles[i] + "_y", H5::PredType::NATIVE_DOUBLE, g,
137 quat_units[i], quat_titles[i] + " (y)", quat_conversions[i]);
138 createDatasetAt(quat_titles[i] + "_z", H5::PredType::NATIVE_DOUBLE, g,
139 quat_units[i], quat_titles[i] + " (z)", quat_conversions[i]);
140 createDatasetAt(quat_titles[i] + "_w", H5::PredType::NATIVE_DOUBLE, g,
141 quat_units[i], quat_titles[i] + " (w)", quat_conversions[i]);
142 }
143
144 for (size_t k = 0; k < mat_titles.size(); k++) {
145 H5::Group* g = groupFor(mat_var_groups[k]);
146 for (int i = 0; i < 3; i++) for (int j = 0; j < 3; j++)
147 createDatasetAt(mat_titles[k] + "_" + std::to_string(i) + std::to_string(j),
148 H5::PredType::NATIVE_DOUBLE, g,
149 mat_units[k],
150 mat_titles[k] + " (" + std::to_string(i) + "," + std::to_string(j) + ")",
151 mat_conversions[k]);
152 }
153
154 file_ready = true;
155 } catch (H5::Exception& e) {
156 std::cerr << "HDF5 Open Failed for: " << deferred_filename
157 << " (" << e.getDetailMsg() << ")" << std::endl;
158 if (file) { delete file; file = nullptr; }
159 } catch (...) {
160 std::cerr << "HDF5 Open Failed for: " << deferred_filename << std::endl;
161 if (file) { delete file; file = nullptr; }
162 }
163 }

206 void report(double t)
207 {
208 // Lazy file creation on first report
209 if (!file_ready) createFile();
210 if (!file) return;
211
212 // Write Time (always at root)
213 appendAt("Time", t, nullptr);
214
215 // Write Doubles
216 for (size_t i=0; i < doubles.size(); i++) {
217 appendAt(titles[i], *doubles[i] * conversions[i], groupFor(var_groups[i]));
218 }
219
220 // Write Vectors
221 for (size_t i=0; i < vectors.size(); i++) {
222 H5::Group* g = groupFor(vec_var_groups[i]);
223 appendAt(vec_titles[i] + "_x", vectors[i]->x * vec_conversions[i], g);
224 appendAt(vec_titles[i] + "_y", vectors[i]->y * vec_conversions[i], g);
225 appendAt(vec_titles[i] + "_z", vectors[i]->z * vec_conversions[i], g);
226 }
227
228 // Write Quaternions
229 for (size_t i=0; i < quats.size(); i++) {
230 H5::Group* g = groupFor(quat_var_groups[i]);
231 appendAt(quat_titles[i] + "_x", quats[i]->q1 * quat_conversions[i], g);
232 appendAt(quat_titles[i] + "_y", quats[i]->q2 * quat_conversions[i], g);
233 appendAt(quat_titles[i] + "_z", quats[i]->q3 * quat_conversions[i], g);
234 appendAt(quat_titles[i] + "_w", quats[i]->q0 * quat_conversions[i], g);
235 }
236
237 // Write Matrices
238 for (size_t k=0; k < matrices.size(); k++) {
239 dsf::util::Mat3& m = *matrices[k];
240 double c = mat_conversions[k];
241 H5::Group* g = groupFor(mat_var_groups[k]);
242 for(int i=0; i<3; i++) for(int j=0; j<3; j++) {
243 appendAt(mat_titles[k] + "_" + std::to_string(i) + std::to_string(j), m[i][j] * c, g);
244 }
245 }
246
247 frame_count++;
248 }

312 void writeRootMetadata()
313 {
314 if (!file) return;
315
316 // Timestamp
317 std::time_t now = std::time(nullptr);
318 std::tm* gmt = std::gmtime(&now);
319 std::ostringstream ts;
320 ts << std::put_time(gmt, "%Y-%m-%dT%H:%M:%SZ");
321 writeStringAttr(*file, "created", ts.str());
322
323 // DSF version (compile-time if available, else "unknown")
324 #ifdef DSF_VERSION
325 writeStringAttr(*file, "dsf_version", DSF_VERSION);
326 #else
327 writeStringAttr(*file, "dsf_version", "dev");
328 #endif
329
330 // XML config
331 if (!meta_xml_file.empty())
332 writeStringAttr(*file, "xml_file", meta_xml_file);
333 if (!meta_xml_content.empty())
334 writeStringAttr(*file, "xml_content", meta_xml_content);
335
336 // Simulation parameters
337 if (meta_dt > 0.0)
338 writeDoubleAttr(*file, "dt", meta_dt);
339 if (meta_tmax > 0.0)
340 writeDoubleAttr(*file, "tmax", meta_tmax);
341 }

352 void createDatasetAt(const std::string& name, const H5::DataType& type,
353 H5::Group* grp,
354 const std::string& units = "",
355 const std::string& long_name = "",
356 double conversion = 1.0)
357 {
358 hsize_t dims[1] = {0};
359 hsize_t maxdims[1] = {H5S_UNLIMITED};
360 H5::DataSpace dataspace(1, dims, maxdims);
361 H5::DSetCreatPropList prop;
362 hsize_t chunk_dims[1] = {1000};
363 prop.setChunk(1, chunk_dims);
364 prop.setDeflate(4); // GZIP compression level 4
365
366 try {
367 H5::DataSet ds;
368 if (grp) ds = grp->createDataSet(name, type, dataspace, prop);
369 else ds = file->createDataSet(name, type, dataspace, prop);
370
371 // Write metadata attributes
372 if (!units.empty())
373 writeStringAttr(ds, "units", units);
374 if (!long_name.empty())
375 writeStringAttr(ds, "long_name", long_name);
376 if (conversion != 1.0)
377 writeDoubleAttr(ds, "conversion_factor", conversion);
378 } catch (...) {}
379 }
380
382 void appendAt(const std::string& name, double value, H5::Group* grp)
383 {
384 try {
385 H5::DataSet ds;
386 if (grp) ds = grp->openDataSet(name);
387 else ds = file->openDataSet(name);
388
389 hsize_t size[1] = {(hsize_t)frame_count + 1};
390 ds.extend(size);
391
392 H5::DataSpace filespace = ds.getSpace();
393 hsize_t offset[1] = {(hsize_t)frame_count};
394 hsize_t dim1[1] = {1};
395 filespace.selectHyperslab(H5S_SELECT_SET, dim1, offset);
396
397 hsize_t memdims[1] = {1};
398 H5::DataSpace memspace(1, memdims);
399 ds.write(&value, H5::PredType::NATIVE_DOUBLE, memspace, filespace);
400 } catch (...) {}
401 }
402 };
