126int EventBus::add(Event e)
127{

129
130 // Resolve any deferred callbacks for this event name
131 for (auto it = deferred_.begin(); it != deferred_.end(); )
132 {
133 if (it->event_name == e.name)
134 {
135 // Chain with existing callback if any
136 auto existing = e.callback;
137 auto deferred_cb = it->callback;
138 if (existing)
139 e.callback = [existing, deferred_cb]() { existing(); deferred_cb(); };

171std::vector<FiredEvent> EventBus::evaluate(double t, double dt)
172{
173 std::vector<FiredEvent> fired_this_step;
174
175 for (int i = 0; i < static_cast<int>(events_.size()); i++)
176 {
177 auto& e = events_[i];
178 auto result = e.check(t, dt);
179 if (result.has_value())
180 {
181 double frac = result.value();
182 double fire_time = t - dt + frac * dt;
183
184 FiredEvent fe;
185 fe.event_id = i;
186 fe.name = e.name;
187 fe.fire_time = fire_time;
188 fe.fraction = frac;
189 fired_this_step.push_back(fe);
190
191 e.fired = true;
192 if (e.one_shot)
193 e.armed = false;
194 }
195 }
196
197 // Sort by crossing time (earliest first)
198 std::sort(fired_this_step.begin(), fired_this_step.end(),
199 [](const FiredEvent& a, const FiredEvent& b) {
200 return a.fraction < b.fraction;
201 });
202
203 // Execute callbacks in order and record history
204 for (auto& fe : fired_this_step)
205 {
206 auto& e = events_[fe.event_id];
207 if (e.callback)
208 {
209 std::cout << "[Event] t=" << fe.fire_time << " '" << fe.name << "'" << std::endl;
210 e.callback();
211 }
212 history_.push_back(fe);
213 }
214
215 return fired_this_step;
216}
