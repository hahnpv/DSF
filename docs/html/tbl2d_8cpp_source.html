148 int Table2d::findBracket(const std::vector<double>& breaks, double val)
149 {
150 int n = (int)breaks.size();
151 if (n < 2) return 0;
152
153 // Clamp to table bounds
154 if (val <= breaks[0]) return 0;
155 if (val >= breaks[n-1]) return n - 2;
156
157 // Binary search
158 int lo = 0, hi = n - 2;
159 while (lo < hi)
160 {
161 int mid = (lo + hi) / 2;
162 if (val < breaks[mid])
163 hi = mid - 1;
164 else if (val >= breaks[mid + 1])
165 lo = mid + 1;
166 else
167 return mid; // breaks[mid] <= val < breaks[mid+1]
168 }
169 return lo;

177 double Table2d::interp(double row_val, double col_val)
178 {
179 // Find bracket indices
180 int ri = findBracket(row_breaks, row_val);
181 int ci = findBracket(col_breaks, col_val);
182
183 // Compute fractional positions within the cell
184 double dr = row_breaks[ri+1] - row_breaks[ri];
185 double dc = col_breaks[ci+1] - col_breaks[ci];
186
187 double t = (dr > 0.0) ? (row_val - row_breaks[ri]) / dr : 0.0;
188 double u = (dc > 0.0) ? (col_val - col_breaks[ci]) / dc : 0.0;
189
190 // Clamp fractions (for extrapolation protection)
191 if (t < 0.0) t = 0.0; if (t > 1.0) t = 1.0;
192 if (u < 0.0) u = 0.0; if (u > 1.0) u = 1.0;
193
194 // Four corners
195 double f00 = grid[ri ][ci ];
196 double f10 = grid[ri+1][ci ];
197 double f01 = grid[ri ][ci+1];
198 double f11 = grid[ri+1][ci+1];
199
200 // Bilinear formula
201 return (1.0 - t) * (1.0 - u) * f00
202 + t * (1.0 - u) * f10
203 + (1.0 - t) * u * f01
204 + t * u * f11;
205 }
