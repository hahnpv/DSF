234 std::vector<double> get_current_values() {
235 std::vector<double> vals;
236 // Pre-allocate to avoid reallocs
237 vals.reserve(doubles.size() + vectors.size()*3 + matrices.size()*9);
238
239 // Doubles
240 for (size_t i = 0; i < doubles.size(); ++i) {
241 vals.push_back(*doubles[i] * conversion[0][i]);
242 }
243 // Vectors
244 for (size_t i = 0; i < vectors.size(); ++i) {
245 double c = conversion[1][i];
246 vals.push_back(vectors[i]->x * c);
247 vals.push_back(vectors[i]->y * c);
248 vals.push_back(vectors[i]->z * c);
249 }
250 // Matrices
251 for (size_t i = 0; i < matrices.size(); ++i) {
252 double c = conversion[2][i];
253 const auto& m = *matrices[i];
254 // Correct access via row vectors (a0, a1, a2)
255 vals.push_back(m.a0.x * c); vals.push_back(m.a0.y * c); vals.push_back(m.a0.z * c);
256 vals.push_back(m.a1.x * c); vals.push_back(m.a1.y * c); vals.push_back(m.a1.z * c);
257 vals.push_back(m.a2.x * c); vals.push_back(m.a2.y * c); vals.push_back(m.a2.z * c);
258 }
259 return vals;
260 }
