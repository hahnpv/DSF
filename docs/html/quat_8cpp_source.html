58 Quaternion Quaternion::quat_mult(const Quaternion& r) const
59 {
60 // Hamilton product: this ⊗ r
61 // [s1,v1] ⊗ [s2,v2] = [s1*s2 - v1·v2, s1*v2 + s2*v1 + v1×v2]
62 return Quaternion(
63 q0*r.q0 - q1*r.q1 - q2*r.q2 - q3*r.q3, // scalar
64 q0*r.q1 + q1*r.q0 + q2*r.q3 - q3*r.q2, // vector-x
65 q0*r.q2 - q1*r.q3 + q2*r.q0 + q3*r.q1, // vector-y
66 q0*r.q3 + q1*r.q2 - q2*r.q1 + q3*r.q0 // vector-z
67 );
68 }
