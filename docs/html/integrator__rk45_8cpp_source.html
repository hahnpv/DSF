82 auto* d = TClassIntegrandDict<Block>::Instance();
83 int n = static_cast<int>(d->x.size());
84 if (n == 0) return;
85
86 // Ensure workspace sized for 7 stages (6 + FSAL)
87 if (d->xdd.size() < 7)
88 d->resize_stages(7);
89
90 double dt_nominal = clock->dt();
91
92 // Initialize adaptive sub-step size on first call
93 if (dt_adapt_ <= 0)
94 dt_adapt_ = dt_nominal;
95
96 // --- Initial derivative evaluation (at current state) ---
97 dsf::util::TFunctor<Block>(root->getChildren(), &Block::update);
98 clock->set(false);
99
100 // Save initial state
101 for (int i = 0; i < n; i++)
102 {
103 d->x0[i] = *d->x[i];
104 d->xdd[0][i] = *d->xd[i]; // k1
105 }
106
107 // --- Sub-step loop: advance by exactly dt_nominal ---
108 double t_covered = 0.0;
109 while (t_covered < dt_nominal - 1e-14)
110 {
111 double t_remaining = dt_nominal - t_covered;
112 double h = std::min(dt_adapt_, t_remaining);
113
114 // Clamp h to exactly finish the macro step
115 if (h > t_remaining * 0.999)
116 h = t_remaining;
117
118 // Save sub-step start state
119 for (int i = 0; i < n; i++)
120 d->x0[i] = *d->x[i];
121
122 // Try a Dormand-Prince step of size h
123 total_steps_++;
124
125 // --- Inline stages (using saved k1 in xdd[0]) ---
126
127 // Stage 2
128 for (int i = 0; i < n; i++)
129 *d->x[i] = d->x0[i] + h * b21 * d->xdd[0][i];

131 for (int i = 0; i < n; i++) d->xdd[1][i] = *d->xd[i];
132
133 // Stage 3

137 for (int i = 0; i < n; i++) d->xdd[2][i] = *d->xd[i];

139 // Stage 4

152
153 // Stage 6
154 for (int i = 0; i < n; i++)
155 *d->x[i] = d->x0[i] + h * (b61 * d->xdd[0][i] + b62 * d->xdd[1][i]
156 + b63 * d->xdd[2][i] + b64 * d->xdd[3][i]
157 + b65 * d->xdd[4][i]);
158 dsf::util::TFunctor<Block>(root->getChildren(), &Block::update);
159 for (int i = 0; i < n; i++) d->xdd[5][i] = *d->xd[i];
160
161 // 5th-order solution
162 for (int i = 0; i < n; i++)
163 *d->x[i] = d->x0[i] + h * (c1 * d->xdd[0][i] + c3 * d->xdd[2][i]
164 + c4 * d->xdd[3][i] + c5 * d->xdd[4][i]
165 + c6 * d->xdd[5][i]);
166
167 // k7 (FSAL) for error estimate
168 dsf::util::TFunctor<Block>(root->getChildren(), &Block::update);
169 for (int i = 0; i < n; i++) d->xdd[6][i] = *d->xd[i];
170
171 // Error estimation
172 double err_max = 0.0;
173 for (int i = 0; i < n; i++)
174 {
175 double err_i = h * std::fabs(e1 * d->xdd[0][i] + e3 * d->xdd[2][i]
176 + e4 * d->xdd[3][i] + e5 * d->xdd[4][i]
177 + e6 * d->xdd[5][i] + e7 * d->xdd[6][i]);
178 double scale = atol_ + rtol_ * std::fabs(*d->x[i]);
179 err_max = std::max(err_max, err_i / scale);
180 }
181
182 if (err_max <= 1.0)
183 {
184 // Accept sub-step
185 t_covered += h;
186
187 // FSAL: k7 of this step becomes k1 of next sub-step
188 for (int i = 0; i < n; i++)
189 d->xdd[0][i] = d->xdd[6][i];
190
191 // Grow dt_adapt for next sub-step
192 double factor = (err_max > 1e-10) ? 0.9 * std::pow(err_max, -0.2) : 5.0;
193 factor = std::min(factor, 5.0);
194 factor = std::max(factor, 0.2);
195 dt_adapt_ = std::min(h * factor, dt_max_);
196 dt_adapt_ = std::max(dt_adapt_, dt_min_);
197 }
198 else
199 {
200 // Reject — restore state, shrink h
201 rejected_steps_++;
202 for (int i = 0; i < n; i++)
203 *d->x[i] = d->x0[i];
204
205 double factor = 0.9 * std::pow(err_max, -0.25);
206 factor = std::max(factor, 0.1);
207 dt_adapt_ = std::max(h * factor, dt_min_);
208
209 // Re-evaluate derivatives at restored state
210 dsf::util::TFunctor<Block>(root->getChildren(), &Block::update);
211 for (int i = 0; i < n; i++)
212 d->xdd[0][i] = *d->xd[i];
213 }
214 }
215
216 // --- Advance clock by nominal dt (2 ticks, same as RK4) ---
217 clock->increment();
218 clock->increment();
219 clock->set(true);
220}
