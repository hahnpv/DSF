86inline MonteCarloCase parse_mc_xml(dsf::xml::xmlnode sim_node)
87{
88 MonteCarloCase mc;
89
90 // Search for <monte_carlo> child node
91 for (int i = 0; i < sim_node.numchild(); i++)
92 {
93 dsf::xml::xmlnode child = sim_node;
94 child.child(i);
95 if (std::string(child.name()) != "monte_carlo") continue;
96
97 mc.n_cases = (int)child.attrAsDouble("n");
98 mc.master_seed = (unsigned int)child.attrAsDouble("seed");
99
100 // Parse <dispersion> children
101 for (int j = 0; j < child.numchild(); j++)
102 {
103 dsf::xml::xmlnode dnode = child;
104 dnode.child(j);
105 if (std::string(dnode.name()) != "dispersion") continue;
106
107 Dispersion d;
108 d.block_id = dnode.attrAsString("block");
109 d.property_name = dnode.attrAsString("property");
110
111 std::string dist = dnode.attrAsString("distribution");
112 if (dist == "uniform")
113 {
114 d.distribution = MCDistribution::UNIFORM;
115 d.min_val = dnode.attrAsDouble("min");
116 d.max_val = dnode.attrAsDouble("max");
117 }
118 else
119 {
120 d.distribution = MCDistribution::GAUSSIAN;
121 d.sigma = dnode.attrAsDouble("sigma");
122 }
123
124 mc.dispersions.push_back(d);
125 }
126 break; // only one <monte_carlo> block
127 }
128
129 return mc;
130}

140inline Block* find_block_by_id(Block* root, const std::string& id)
141{
142 if (!root) return nullptr;
143
144 // Check this block's name
145 if (root->getName() == id)
146 return root;
147
148 // Search children recursively
149 for (Block* child : root->getChildren())
150 {
151 Block* found = find_block_by_id(child, id);
152 if (found) return found;
153 }
154 return nullptr;
155}

168inline bool lookup_property_offset(Block* block, const std::string& property_name, size_t& offset)
169{
170 std::string class_name = dsf::util::demangle(typeid(*block).name());
171
172 auto* dict = TClassDict<Block>::Instance();
173 for (auto* entry : dict->classDictPtr)
174 {
175 if (entry->name() == class_name)
176 {
177 for (const auto& prop : entry->getProperties())
178 {
179 if (prop.name == property_name && prop.offset > 0)
180 {
181 offset = prop.offset;
182 return true;
183 }
184 }
185 std::cout << "[MC] Property '" << property_name
186 << "' not found (or not BIND'd) in class '" << class_name << "'" << std::endl;
187 return false;
188 }
189 }
190 std::cout << "[MC] Class '" << class_name << "' not found in TClassDict" << std::endl;
191 return false;
192}

209inline void apply_dispersions(Block* root, MonteCarloCase& mc)
210{
211 if (!mc.is_mc() || mc.dispersions.empty()) return;
212
213 // Seed RNG for this case
214 dsf::util::set_seed(mc.case_seed);
215
216 std::cout << "\n[MC] Case " << mc.case_id << "/" << mc.n_cases
217 << " (seed=" << mc.case_seed << ")" << std::endl;
218
219 for (auto& d : mc.dispersions)
220 {
221 // Find the block in the tree
222 Block* block = find_block_by_id(root, d.block_id);
223 if (!block)
224 {
225 std::cout << "[MC] WARNING: Block '" << d.block_id << "' not found" << std::endl;
226 continue;
227 }
228
229 // Look up property offset
230 size_t offset = 0;
231 if (!lookup_property_offset(block, d.property_name, offset))
232 continue;
233
234 // Read nominal value (set by configure())
235 double* member_ptr = reinterpret_cast<double*>(
236 reinterpret_cast<char*>(block) + offset);
237 d.nominal = *member_ptr;
238
239 // Draw dispersed value
240 switch (d.distribution)
241 {
242 case MCDistribution::GAUSSIAN:
243 d.drawn = dsf::util::get_gauss(d.nominal, d.sigma);
244 d.n_sigma = (d.sigma > 0.0) ? (d.drawn - d.nominal) / d.sigma : 0.0;
245 break;
246
247 case MCDistribution::UNIFORM:
248 d.drawn = dsf::util::getUniform(d.min_val, d.max_val);
249 d.n_sigma = 0.0; // not meaningful for uniform
250 break;
251 }
252
253 // Write dispersed value
254 *member_ptr = d.drawn;
255
256 std::cout << "[MC] " << d.block_id << "." << d.property_name
257 << ": nominal=" << d.nominal << " drawn=" << d.drawn;
258 if (d.distribution == MCDistribution::GAUSSIAN)
259 std::cout << " (" << d.n_sigma << " sigma)";
260 std::cout << std::endl;
261 }
262 std::cout << std::endl;
263}
